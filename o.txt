#jobseq
#include <stdio.h>

typedef struct task {
    int task_id, deadline, profit;
} task;

task tasklist[100];

int highestDeadline(int n) {
    int max = 0;
    if (n == 0) return 0;

    for (int i = 0; i < n; i++) {
        max = (tasklist[i].deadline > max) ? tasklist[i].deadline : max;
    }

    return max;
}

void collectTasks(int n) {
    int profit, id, deadline;
    for (int i = 0; i < n; i++) {
        scanf("%d %d %d", &id, &deadline, &profit);
        tasklist[i].task_id = id;
        tasklist[i].profit = profit;
        tasklist[i].deadline = deadline;
    }
}

void sortTasks(int n) {
    task temp;
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - 1 - i; j++) {
            if (tasklist[j].profit < tasklist[j + 1].profit) {
                temp = tasklist[j];
                tasklist[j] = tasklist[j + 1];
                tasklist[j + 1] = temp;
            }
        }
    }
}

void printSeq(int a[], int n) {
    printf("Scheduled tasks: ");
    for (int i = 0; i < n; i++) {
        if (a[i] != -1) printf("%d ", a[i]);
    }
}

void schedule(int n) {
    int maxdeadline = highestDeadline(n);
    int gant[100], tindex, max_profit = 0;
    
    for (int i = 0; i < maxdeadline; i++) gant[i] = -1;
    
    sortTasks(n);
    
    for (int i = 0; i < n; i++) {
        tindex = tasklist[i].deadline - 1;
        while (1) {
            if (tindex == 0 && gant[tindex] != -1) break;
            
            if (gant[tindex] != -1 && tindex != 0) {
                tindex--;
            } else {
                gant[tindex] = tasklist[i].task_id;
                max_profit += tasklist[i].profit;
                break;
            }
        }
    }
    
    printSeq(gant, maxdeadline);
    printf("\nMaximum Profit: %d\n", max_profit);
    return;
}

int main(void) {
    int no_of_tasks;
    scanf("%d", &no_of_tasks);
    collectTasks(no_of_tasks);
    schedule(no_of_tasks);
    return 0;
}








#rabinkarp
#include <stdio.h>
#include <string.h>

#define d 256
#define q 101

void rabinKarp(char text[], char pattern[]) {
    int n = strlen(text);
    int m = strlen(pattern);
    int i, j;
    int p = 0;
    int t = 0;
    int h = 1;
    int found = 0;

    for (i = 0; i < m - 1; i++) {
        h = (h * d) % q;
    }

    for (i = 0; i < m; i++) {
        p = (d * p + pattern[i]) % q;
        t = (d * t + text[i]) % q;
    }

    for (i = 0; i <= n - m; i++) {
        if (p == t) {
            for (j = 0; j < m; j++) {
                if (text[i + j] != pattern[j]) {
                    break;
                }
            }
            if (j == m) {
                printf("Pattern found at index: %d\n", i);
                found = 1;
            }
        }

        if (i < n - m) {
            t = (d * (t - text[i] * h) + text[i + m]) % q;

            if (t < 0) {
                t = t + q;
            }
        }
    }

    if (found == 0) {
        printf("Pattern not found\n");
    }
}

int main() {
    char text[1000];
    char pattern[1000];

    scanf("%s", text);
    scanf("%s", pattern);

    rabinKarp(text, pattern);

    return 0;
}










#coin
#include <stdio.h>
#include <limits.h>

int main() {
    int n, target;

    if (scanf("%d", &n) != 1) {
        return 0;
    }

    int coins[n];

    for (int i = 0; i < n; i++) {
        scanf("%d", &coins[i]);
    }

    if (scanf("%d", &target) != 1) {
        return 0;
    }

    int dp[target + 1];
    dp[0] = 0;

    for (int i = 1; i <= target; i++) {
        dp[i] = INT_MAX;
    }

    for (int i = 1; i <= target; i++) {
        for (int j = 0; j < n; j++) {
            
            if (coins[j] <= i) {
                int res = dp[i - coins[j]];
                
                if (res != INT_MAX && res + 1 < dp[i]) {
                    dp[i] = res + 1;
                }
            }
        }
    }

    if (dp[target] == INT_MAX) {
        printf("-1\n");
    }
    else {
        printf("Minimum coins required: %d\n", dp[target]);
    }

    return 0;
}

            for (int k = i; k <= j - 1; k++) {

                    m[i][j] = q;
                }

#include <stdio.h>
#include <string.h>

#define MAX 1000

void findLCS(char *X, char *Y) {
    int m = strlen(X);
    int n = strlen(Y);
    int L[m+1][n+1];
    
    // Build the DP table
    for (int i = 0; i <= m; i++) {
        for (int j = 0; j <= n; j++) {
            
            // Fix: Initialize the entire 0th row and 0th column to 0
            if (i == 0 || j == 0) {
                L[i][j] = 0;
            }
            // Fix: X and Y are 1D arrays, so use single indices
            else if (X[i-1] == Y[j-1]) {
                L[i][j] = L[i-1][j-1] + 1;
            }
            else {
                // Fix: Added the missing semicolon
                L[i][j] = (L[i-1][j] > L[i][j-1]) ? L[i-1][j] : L[i][j-1];
            }
        }
    }
    
    int index = L[m][n];
    // Fix: Corrected printf syntax and added newline
    printf("LCS length is %d\n", index);
    
    char lcs[index + 1];
    // Fix: Corrected null terminator syntax (backslash and single quotes)
    lcs[index] = '\0';
    
    int i = m, j = n;
    
    // Trace back to find the characters
    while (i > 0 && j > 0) {
        // Fix: X and Y are 1D arrays
        if (X[i-1] == Y[j-1]) {
            lcs[index-1] = X[i-1];
            i--;
            j--;
            index--;
        }
        else if (L[i-1][j] > L[i][j-1]) {
            i--;
        }
        else {
            j--;
        }
    }
    
    // Fix: Corrected printf syntax and added newline
    printf("LCS is %s\n", lcs);
}

int main() {
    char X[MAX], Y[MAX];
     
    if (scanf("%s", X) == 1 && scanf("%s", Y) == 1) {
        findLCS(X, Y);
    }
    return 0;
}

#include<stdio.h>

long long recursiveFib(int n){

    if (n<=1){
        return n;
    }

    return recursiveFib(n-1) + recursiveFib(n-2);
}

long long iterativeFib(int n){

    if (n <= 1){
        return n;
    }

    long long current = 1;
    long long prev = 0;
    long long next;

    for(int i=2;i <=n;i++){

        next = prev + current;
        prev = current;
        current = next;
    }

    return current;
}

int main(){

    int n;

    if(scanf("%d", &n) != 1){
        return 0;
    }

    long long recursiveRes = recursiveFib(n);
    long long iterativeRes = iterativeFib(n);

    printf("Recursive Fib (%d):%lld\n", n, recursiveRes);
    printf("Iterative Fib (%d):%lld\n", n, iterativeRes);

}
